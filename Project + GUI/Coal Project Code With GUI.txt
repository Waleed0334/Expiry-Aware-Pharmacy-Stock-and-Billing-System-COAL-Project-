; Expiry-Aware Pharmacy Stock and Billing System  -  GUI VERSION (final)
; x86 MASM (32-bit), plain Win32 API (kernel32 / user32 / gdi32)
;
; Irvine32 only works in a console window, so this version talks to
; Windows directly. The pharmacy logic (arrays, batches, expiry,
; earliest-expiry-first selling, bill, reports, file) is the same as
; the console program. Only the screen and keyboard part is new.
;
; Screen layout
;   left   : 11 menu buttons (owner-drawn: Expiry Loss = red,
;            Reorder = yellow, like the colours in the proposal)
;   top    : up to 3 labelled input boxes + OK button (+ Finish Bill)
;   centre : black output list (owner-drawn, so lines can be coloured)
;
; Build (Visual Studio 2022, x86):
;   - add this file, item type = Microsoft Macro Assembler
;   - Linker > System > SubSystem        : Windows
;   - Linker > Advanced > Entry Point    : start
;   - Linker > Advanced > Image Has Safe Exception Handlers : No
;   - Irvine32 is NOT needed for this file
;
; Password : pharma123
; ====================================================================

; ====================================================================
; PROGRAM START
; ====================================================================
start:
    invoke GetModuleHandleA, 0
    mov  hInst, eax

    call LoadDataFromFile               ; silent, the window reports the result

    mov  wc.cbSize, SIZEOF WNDCLASSEXA
    mov  wc.style, 3                    ; CS_HREDRAW or CS_VREDRAW
    mov  wc.lpfnWndProc, OFFSET WndProc
    mov  wc.cbClsExtra, 0
    mov  wc.cbWndExtra, 0
    mov  eax, hInst
    mov  wc.hInstance, eax
    mov  wc.hIcon, 0
    mov  wc.hIconSm, 0
    mov  wc.lpszMenuName, 0
    invoke LoadCursorA, 0, IDC_ARROW
    mov  wc.hCursor, eax
    mov  wc.hbrBackground, 16           ; COLOR_BTNFACE + 1
    mov  wc.lpszClassName, OFFSET clsAppName
    invoke RegisterClassExA, ADDR wc

    invoke CreateWindowExA, 0, ADDR clsAppName, ADDR windowTitle, \
           WS_MAINWINDOW or WS_VISIBLE, 40, 20, 1010, 700, 0, 0, hInst, 0

MessageLoop:
    invoke GetMessageA, ADDR msg, 0, 0, 0
    test eax, eax
    jz   LoopEnd
    invoke IsDialogMessageA, hMain, ADDR msg    ; gives Tab and Enter keys
    test eax, eax
    jnz  MessageLoop
    invoke TranslateMessage, ADDR msg
    invoke DispatchMessageA, ADDR msg
    jmp  MessageLoop
LoopEnd:
    invoke ExitProcess, msg.wParam


; ====================================================================
; WINDOW PROCEDURE : Windows calls this for every event
; ====================================================================
WndProc PROC USES ebx esi edi, hWnd:DWORD, uMsg:DWORD, wParam:DWORD, lParam:DWORD
    mov  eax, uMsg
    cmp  eax, WM_CREATE
    je   OnCreate
    cmp  eax, WM_COMMAND
    je   OnCommand
    cmp  eax, WM_DRAWITEM
    je   OnDrawItem
    cmp  eax, WM_MEASUREITEM
    je   OnMeasureItem
    cmp  eax, WM_CTLCOLORLISTBOX
    je   OnColorList
    cmp  eax, WM_CLOSE
    je   OnClose
    cmp  eax, WM_DESTROY
    je   OnDestroy
    invoke DefWindowProcA, hWnd, uMsg, wParam, lParam
    ret

; ---------- window is being created: build every control
OnCreate:
    mov  eax, hWnd
    mov  hMain, eax
    invoke GetStockObject, DEFAULT_GUI_FONT
    mov  hGuiFont, eax
    invoke GetStockObject, ANSI_FIXED_FONT
    mov  hFixedFont, eax
    invoke CreateSolidBrush, 0
    mov  hBlackBrush, eax
    invoke GetSysColorBrush, COLOR_BTNFACE_ID   ; system brush, never deleted
    mov  hBrushFace, eax
    invoke CreateSolidBrush, BTN_RED
    mov  hBrushRed, eax
    invoke CreateSolidBrush, BTN_YELLOW
    mov  hBrushYellow, eax

    ; 11 owner-drawn menu buttons, ids 101..111
    xor  esi, esi
CreateBtnLoop:
    cmp  esi, 11
    jge  CreateBtnDone
    imul ebx, esi, 38
    add  ebx, 10                        ; y position
    lea  edi, [esi + 101]               ; control id
    invoke CreateWindowExA, 0, ADDR clsButton, btnTexts[esi*4], \
           WS_CHILD or WS_VISIBLE or WS_TABSTOP or BS_OWNERDRAW, 10, ebx, 215, 32, hWnd, edi, hInst, 0
    mov  hBtns[esi*4], eax
    invoke ApplyFont, eax
    inc  esi
    jmp  CreateBtnLoop
CreateBtnDone:

    invoke ApplyFont, eax
    invoke SendMessageA, hEdit1, EM_LIMITTEXT, 30, 0
    invoke CreateWindowExA, WS_EX_CLIENTEDGE, ADDR clsEdit, ADDR emptyStr, \

; Copies a 0-terminated string
StrCopy PROC USES eax esi edi, pSrc:DWORD, pDst:DWORD
    mov  esi, pSrc
    mov  edi, pDst
scpNext:
    mov  al, [esi]
    mov  [edi], al
    inc  esi
    inc  edi
    test al, al
    jnz  scpNext
    ret
StrCopy ENDP

; Compares two strings. Returns EAX = -1 (A < B), 0 (equal), 1 (A > B).
; fold = 1 ignores upper/lower case.
StrCompare PROC USES ebx esi edi, pA:DWORD, pB:DWORD, fold:DWORD
    mov  esi, pA
    mov  edi, pB
scLoop:
    movzx eax, BYTE PTR [esi]
    movzx ebx, BYTE PTR [edi]
    cmp  fold, 0
    je   scCompare
    cmp  al, 'a'                        ; turn lower case into upper case
    jb   scFoldB
    cmp  al, 'z'
    ja   scFoldB
    sub  al, 20h
scFoldB:
    cmp  bl, 'a'
    jb   scCompare
    cmp  bl, 'z'
    ja   scCompare
    sub  bl, 20h
scCompare:
    cmp  eax, ebx
    jb   scLess
    ja   scGreater
    test eax, eax                       ; equal and both ended
    jz   scEqual
    inc  esi
    inc  edi
    jmp  scLoop
scLess:
    mov  eax, -1
    ret
scGreater:
    mov  eax, 1
    ret
scEqual:
    xor  eax, eax
    ret
StrCompare ENDP

; Reads a whole number from an edit box. Digits only, at most 9 of them.
; Spaces before or after the number are ignored; anything else is rejected.
; Returns EAX = value, EDX = 1 if it was a valid number, 0 if not.
GetNum PROC USES ebx ecx esi, hEd:DWORD
    invoke GetWindowTextA, hEd, ADDR numBuf, 31
    mov  esi, OFFSET numBuf
    xor  eax, eax                       ; value
    xor  ecx, ecx                       ; digits seen
    xor  ebx, ebx                       ; 1 once a space came after the digits
gnNext:
    movzx edx, BYTE PTR [esi]
    test edx, edx
    jz   gnEnd
    cmp  edx, ' '
    jne  gnDigit
    test ecx, ecx
    jz   gnSkip                         ; leading space is ignored
    mov  ebx, 1                         ; trailing space: no more digits allowed
    jmp  gnSkip
gnDigit:
    cmp  edx, '0'
    jb   gnBad
    cmp  edx, '9'
    ja   gnBad
    test ebx, ebx
    jnz  gnBad                          ; digit after a space (like "1 2")
    sub  edx, '0'
    imul eax, eax, 10
    add  eax, edx
    inc  ecx
    cmp  ecx, 9
    ja   gnBad
gnSkip:
    inc  esi
    jmp  gnNext
gnEnd:
    test ecx, ecx
    jz   gnBad
    mov  edx, 1
    ret
gnBad:
    xor  eax, eax
    xor  edx, edx
    ret
GetNum ENDP

; Checks a YYYYMMDD date (2000-01-01 .. 2100-12-31, leap years respected).
; Returns EAX = 1 (valid) or 0 (invalid).
IsValidDate PROC USES ebx ecx edx, dateVal:DWORD
    mov  eax, dateVal
    cmp  eax, 20000101
    jl   ivBad
    cmp  eax, 21001231
    jg   ivBad
    xor  edx, edx
    mov  ecx, 10000
    div  ecx                            ; EAX = year, EDX = MMDD
    mov  ebx, eax                       ; EBX = year
    mov  eax, edx
    xor  edx, edx
    mov  ecx, 100
    div  ecx                            ; EAX = month, EDX = day
    cmp  eax, 1
    jl   ivBad
    cmp  eax, 12
    jg   ivBad
    cmp  edx, 1
    jl   ivBad
    movzx ecx, BYTE PTR monthDays[eax - 1]
    cmp  edx, ecx
    jg   ivBad
    cmp  eax, 2                         ; only 29 February needs the leap test
    jne  ivGood
    cmp  edx, 29
    jne  ivGood
    mov  eax, ebx                       ; year divisible by 4 ?
    xor  edx, edx
    mov  ecx, 4
    div  ecx
    test edx, edx
    jnz  ivBad
    mov  eax, ebx                       ; ...and not by 100 -> leap year
    xor  edx, edx
    mov  ecx, 100
    div  ecx
    test edx, edx
    jnz  ivGood
    mov  eax, ebx                       ; divisible by 100: leap only if by 400
    xor  edx, edx
    mov  ecx, 400
    div  ecx
    test edx, edx
    jnz  ivBad
ivGood:
    mov  eax, 1
    ret
ivBad:
    xor  eax, eax
    ret
IsValidDate ENDP

; YYYYMMDD -> simple day number (every month = 31 days)
DateToDays PROC USES ebx ecx edx, dateVal:DWORD
    mov  eax, dateVal
    xor  edx, edx
    mov  ebx, 10000
    div  ebx                            ; EAX = year, EDX = MMDD
    imul ecx, eax, 372                  ; 12 months * 31 days
    mov  eax, edx
    xor  edx, edx
    mov  ebx, 100
    div  ebx                            ; EAX = month, EDX = day
    imul eax, eax, 31
    add  eax, edx
    add  eax, ecx
    ret
DateToDays ENDP

; Returns EAX = 0 usable, 1 expires within 30 days, 2 expired
GetBatchStatus PROC USES ebx, expiry:DWORD
    mov  eax, expiry
    cmp  eax, todayDate
    jl   gsExpired
    invoke DateToDays, eax
    mov  ebx, eax
    invoke DateToDays, todayDate
    sub  ebx, eax                       ; days left
    cmp  ebx, SOON_DAYS
    jle  gsSoon
    mov  eax, STATUS_OK
    ret
gsSoon:
    mov  eax, STATUS_SOON
    ret
gsExpired:
    mov  eax, STATUS_EXPIRED
    ret
GetBatchStatus ENDP

; Returns EAX = index of the medicine with this name (ignores case), or -1

fnLoop:
    cmp  esi, medCount
    jge  fnNone
    mov  eax, esi
    call GetNameAddr                    ; EDX = name of medicine ESI
    invoke StrCompare, edx, pName, 1
    test eax, eax
    jz   fnFound
    inc  esi
    jmp  fnLoop

    mov  eax, -1

fnFound:
    mov  eax, esi
    ret
FindMedicineByName ENDP

; Returns EAX = total quantity in the non-expired batches of a medicine
GetUsableStock PROC USES ebx ecx edx esi, medIdx:DWORD
    xor  ebx, ebx
    xor  esi, esi
guLoop:
    cmp  esi, batCount
    jge  guDone
    mov  eax, medIdx
    cmp  batMed[esi*4], eax
    jne  guNext
    invoke GetBatchStatus, batExpiry[esi*4]
    cmp  eax, STATUS_EXPIRED
    je   guNext
    add  ebx, batQty[esi*4]
guNext:
    inc  esi
    jmp  guLoop
guDone:
    mov  eax, ebx
    ret
GetUsableStock ENDP

; Returns EAX = batch index with the earliest expiry date that is not expired
; and not empty, or -1 if there is none
FindEarliestBatch PROC USES ebx ecx edx esi edi, medIdx:DWORD
    mov  edi, -1                        ; best batch so far
    mov  edx, 99999999                  ; its expiry date
    xor  esi, esi
feLoop:
    cmp  esi, batCount
    jge  feDone
    mov  eax, medIdx
    cmp  batMed[esi*4], eax
    jne  feNext
    cmp  batQty[esi*4], 0
    je   feNext
    mov  eax, batExpiry[esi*4]
    cmp  eax, todayDate                 ; expired batches are ignored
    jl   feNext
    cmp  eax, edx
    jge  feNext
    mov  edx, eax
    mov  edi, esi
feNext:
    inc  esi
    jmp  feLoop
feDone:
    mov  eax, edi
    ret
FindEarliestBatch ENDP

; Removes one batch by shifting the later ones down
RemoveBatchAt PROC USES eax ebx ecx esi, idx:DWORD
    mov  esi, idx
rbLoop:
    mov  ebx, esi
    inc  ebx
    cmp  ebx, batCount
    jge  rbDone
    mov  ecx, batMed[ebx*4]
    mov  batMed[esi*4], ecx
    mov  ecx, batQty[ebx*4]
    mov  batQty[esi*4], ecx
    mov  ecx, batExpiry[ebx*4]
    mov  batExpiry[esi*4], ecx
    inc  esi
    jmp  rbLoop
rbDone:
    dec  batCount
    ret
RemoveBatchAt ENDP

; Deletes every batch whose quantity is 0
CleanEmptyBatches PROC USES eax esi
    xor  esi, esi
ceLoop:
    cmp  esi, batCount
    jge  ceDone
    cmp  batQty[esi*4], 0
    jne  ceKeep
    invoke RemoveBatchAt, esi
    jmp  ceLoop                         ; next batch slid into this slot
ceKeep:
    inc  esi
    jmp  ceLoop
ceDone:
    ret
CleanEmptyBatches ENDP

; Removes one medicine record by shifting the later ones down
RemoveMedicineAt PROC USES eax ebx ecx edx esi edi, idx:DWORD
    mov  esi, idx
rmLoop:
    mov  ebx, esi
    inc  ebx
    cmp  ebx, medCount
    jge  rmDone
    mov  ecx, medID[ebx*4]
    mov  medID[esi*4], ecx
    mov  ecx, medPrice[ebx*4]
    mov  medPrice[esi*4], ecx
    mov  ecx, medSold[ebx*4]
    mov  medSold[esi*4], ecx
    mov  eax, ebx
    call GetNameAddr
    mov  edi, edx                       ; source name
    mov  eax, esi
    call GetNameAddr                    ; EDX = target name
    invoke StrCopy, edi, edx
    inc  esi
    jmp  rmLoop
rmDone:
    dec  medCount
    ret
RemoveMedicineAt ENDP

; Deletes a medicine and all of its batches. Batches of later medicines
; get their medicine index lowered by one.
DeleteMedicine PROC USES eax esi edi, idx:DWORD
    mov  edi, idx
    xor  esi, esi
dmBatchLoop:
    cmp  esi, batCount
    jge  dmBatchesDone
    cmp  batMed[esi*4], edi
    jne  dmKeepBatch
    invoke RemoveBatchAt, esi
    jmp  dmBatchLoop
dmKeepBatch:
    inc  esi
    jmp  dmBatchLoop
dmBatchesDone:
    invoke RemoveMedicineAt, edi
    xor  esi, esi
dmFixLoop:
    cmp  esi, batCount
    jge  dmFixDone
    cmp  batMed[esi*4], edi
    jle  dmSkipFix
    dec  batMed[esi*4]
dmSkipFix:
    inc  esi
    jmp  dmFixLoop
dmFixDone:
    ret
DeleteMedicine ENDP

; Returns EAX = -1, 0 or 1 comparing the names of medicine i and j
CompareMedNames PROC USES ebx edx, i:DWORD, j:DWORD
    mov  eax, i
    call GetNameAddr
    mov  ebx, edx
    mov  eax, j
    call GetNameAddr
    invoke StrCompare, ebx, edx, 1
    ret
CompareMedNames ENDP

; Swaps medicine i and j completely and fixes the batch links
SwapMedicines PROC USES eax ebx ecx edx esi edi, i:DWORD, j:DWORD
    mov  esi, i
    mov  edi, j

    mov  eax, medID[esi*4]
    xchg eax, medID[edi*4]
    mov  medID[esi*4], eax

    mov  eax, medPrice[esi*4]
    xchg eax, medPrice[edi*4]
    mov  medPrice[esi*4], eax

    mov  eax, medSold[esi*4]
    xchg eax, medSold[edi*4]
    mov  medSold[esi*4], eax

    mov  eax, esi                       ; swap the names through tempName
    call GetNameAddr
    mov  ebx, edx
    mov  eax, edi
    call GetNameAddr
    invoke StrCopy, ebx, ADDR tempName
    invoke StrCopy, edx, ebx
    invoke StrCopy, ADDR tempName, edx

    xor  ecx, ecx                       ; batches follow their medicine
swFix:
    cmp  ecx, batCount
    jge  swDone
    mov  eax, batMed[ecx*4]
    cmp  eax, esi
    je   swToSecond
    cmp  eax, edi
    je   swToFirst
    jmp  swNext
swToSecond:
    mov  batMed[ecx*4], edi
    jmp  swNext
swToFirst:
    mov  batMed[ecx*4], esi
swNext:
    inc  ecx
    jmp  swFix
swDone:
    ret
SwapMedicines ENDP

; key = 1 sort by name (A to Z), key = 2 sort by price (low to high)
SortMedicines PROC USES ebx ecx edx esi edi, key:DWORD
    mov  ecx, medCount
    dec  ecx                            ; number of passes
smOuter:
    push ecx
    xor  esi, esi
smInner:
    mov  edi, esi
    inc  edi
    cmp  edi, medCount
    jge  smInnerDone
    cmp  key, 1
    je   smByName
    mov  eax, medPrice[esi*4]
    cmp  eax, medPrice[edi*4]
    jle  smNoSwap
    jmp  smDoSwap
smByName:
    invoke CompareMedNames, esi, edi
    cmp  eax, 0
    jle  smNoSwap
smDoSwap:
    invoke SwapMedicines, esi, edi
smNoSwap:
    inc  esi
    jmp  smInner
smInnerDone:
    pop  ecx
    dec  ecx
    jnz  smOuter
    ret
SortMedicines ENDP

    cmp  medCount, 2

; Takes qty units of a medicine from its batches, earliest expiry first.
; Each batch used is reported in the output list.
TakeStockFromBatches PROC USES ebx esi edi, medIdx:DWORD, qty:DWORD
    mov  edi, qty                       ; EDI = units still to take
tsLoop:
    test edi, edi
    jz   tsDone
    invoke FindEarliestBatch, medIdx
    cmp  eax, -1
    je   tsDone
    mov  esi, eax                       ; ESI = chosen batch
    mov  ebx, batQty[esi*4]             ; EBX = what this batch has
    cmp  ebx, edi
    jle  tsTake                         ; batch has less or equal: take it all
    mov  ebx, edi                       ; batch has more: take only what we need
tsTake:
    sub  batQty[esi*4], ebx
    sub  edi, ebx
    invoke wsprintfA, ADDR lineBuf, ADDR fmtTook, ebx, batExpiry[esi*4]
    invoke PutLine, ADDR lineBuf, CLR_GRAY
    jmp  tsLoop
tsDone:
    call CleanEmptyBatches
    ret
TakeStockFromBatches ENDP

; Adds one item (ID + quantity) to the open bill
OkSell PROC USES ebx esi edi
    cmp  billItemCount, MAX_BILL_ITEMS
    jl   osRoom
    invoke ShowInfo, ADDR msgBillFull
    call FinishBill
    ret
osRoom:
    invoke GetRangeField, hEdit1, 1, MAX_ID, ADDR msgBadId
    test edx, edx
    jz   osEnd
    invoke FindMedicineById, eax
    cmp  eax, -1
    jne  osFound
    invoke ShowError, ADDR msgNotFound
    invoke SetFocus, hEdit1
    ret
osFound:
    mov  ebx, eax                       ; EBX = medicine index
    invoke GetRangeField, hEdit2, 1, MAX_QTY, ADDR msgBadQty
    test edx, edx
    jz   osEnd
    mov  esi, eax                       ; ESI = quantity wanted
    invoke GetUsableStock, ebx
    mov  edi, eax                       ; EDI = usable stock
    test edi, edi
    jnz  osHasStock
    invoke ShowError, ADDR msgOutOfStock
    ret
osHasStock:
    cmp  esi, edi
    jle  osEnough
    invoke wsprintfA, ADDR lineBuf, ADDR fmtNotEnough, edi
    invoke ShowError, ADDR lineBuf
    ret
osEnough:
    mov  eax, esi                       ; would the bill become too large?
    imul eax, medPrice[ebx*4]
    add  eax, billSubtotal
    cmp  eax, MAX_BILL_TOTAL
    jbe  osCapOk
    invoke ShowError, ADDR msgBillTooBig
    ret
osCapOk:
    cmp  billItemCount, 0               ; first item of a new bill: fresh list
    jne  osTake
    call ClearList
osTake:
    invoke TakeStockFromBatches, ebx, esi

    mov  eax, esi                       ; amount = quantity * price
    imul eax, medPrice[ebx*4]
    add  billSubtotal, eax
    mov  edi, billItemCount
    mov  billItemMed[edi*4], ebx
    mov  billItemQty[edi*4], esi
    mov  billItemAmount[edi*4], eax
    inc  billItemCount

    add  medSold[ebx*4], esi            ; used for "fast sellers first"
    add  itemsSold, esi
    call AutoSave

    mov  eax, ebx
    call GetNameAddr
    invoke wsprintfA, ADDR lineBuf, ADDR fmtItem, edx, esi, billItemAmount[edi*4]
    invoke PutLine, ADDR lineBuf, CLR_GREEN
    call ResetInputs
osEnd:
    ret
OkSell ENDP

; Discount: 5% (= subtotal / 20) when the subtotal is Rs. 1000 or more
CalculateBill PROC USES eax ebx edx
    mov  billDiscount, 0
    mov  eax, billSubtotal
    cmp  eax, DISCOUNT_LIMIT
    jb   cbNoDiscount                   ; unsigned compare
    xor  edx, edx                       ; DIV uses EDX:EAX
    mov  ebx, 20
    div  ebx                            ; EAX = subtotal / 20 = 5 percent
    mov  billDiscount, eax
cbNoDiscount:
    mov  eax, billSubtotal
    sub  eax, billDiscount
    mov  billTotal, eax
    ret
CalculateBill ENDP

; ====================================================================
; 8. EXPIRY LOSS REPORT
; ====================================================================
ExpiryLossReport PROC USES eax ebx ecx edx esi edi
    mov  curMode, MODE_NONE
    invoke ShowForm, ADDR infoLoss, ADDR emptyStr, ADDR emptyStr, ADDR emptyStr, 0
    call ClearList
    mov  expiredLoss, 0
    mov  soonLoss, 0
    mov  lossRows, 0
    invoke PutLine, ADDR msgLossHead, CLR_CYAN

    xor  esi, esi                       ; ESI = batch index
elLoop:
    cmp  esi, batCount
    jge  elDone
    invoke GetBatchStatus, batExpiry[esi*4]
    cmp  eax, STATUS_OK
    je   elNext
    mov  ebx, eax                       ; EBX = status

    mov  edi, batMed[esi*4]             ; EDI = medicine index
    mov  eax, batQty[esi*4]
    imul eax, medPrice[edi*4]           ; EAX = quantity * price
    mov  tmpValue, eax
    cmp  ebx, STATUS_EXPIRED
    jne  elIsSoon
    add  expiredLoss, eax
    jnc  elExpOk
    mov  expiredLoss, 0FFFFFFFFh        ; never wrap around: stay at the maximum
elExpOk:
    mov  tmpColor, CLR_RED
    mov  tmpTag, OFFSET tagExpired
    jmp  elPrint
elIsSoon:
    add  soonLoss, eax
    jnc  elSoonOk
    mov  soonLoss, 0FFFFFFFFh
elSoonOk:
    mov  tmpColor, CLR_YELLOW
    mov  tmpTag, OFFSET tagSoonShort
elPrint:
    mov  eax, edi
    call GetNameAddr                    ; EDX = medicine name
    invoke wsprintfA, ADDR lineBuf, ADDR fmtLoss, tmpTag, edx, batQty[esi*4], batExpiry[esi*4], tmpValue
    invoke PutLine, ADDR lineBuf, tmpColor
    inc  lossRows
elNext:
    inc  esi
    jmp  elLoop
elDone:
    cmp  lossRows, 0
    jne  elTotals
    invoke PutLine, ADDR msgNoRisk, CLR_GREEN
    ret
elTotals:
    invoke PutLine, ADDR emptyStr, CLR_WHITE
    invoke wsprintfA, ADDR lineBuf, ADDR fmtExpiredTot, expiredLoss
    invoke PutLine, ADDR lineBuf, CLR_RED
    invoke wsprintfA, ADDR lineBuf, ADDR fmtSoonTot, soonLoss
    invoke PutLine, ADDR lineBuf, CLR_YELLOW
    mov  eax, expiredLoss
    add  eax, soonLoss
    jnc  elRiskOk
    mov  eax, 0FFFFFFFFh
elRiskOk:
    invoke wsprintfA, ADDR lineBuf, ADDR fmtRiskTot, eax
    invoke PutLine, ADDR lineBuf, CLR_CYAN
    ret
ExpiryLossReport ENDP


; ====================================================================
; 9. REORDER SUGGESTION
; ====================================================================
ReorderSuggestion PROC USES eax ebx ecx edx esi edi
    mov  curMode, MODE_NONE
    invoke ShowForm, ADDR infoReorder, ADDR emptyStr, ADDR emptyStr, ADDR emptyStr, 0
    call ClearList
    invoke PutLine, ADDR msgReorderHead, CLR_YELLOW

    mov  lowCount, 0                    ; step 1: collect low-stock medicines
    xor  esi, esi
rsCollect:
    cmp  esi, medCount
    jge  rsCollected
    invoke GetUsableStock, esi
    cmp  eax, LOW_STOCK_LIMIT
    jge  rsCollectNext
    mov  edi, lowCount
    mov  lowList[edi*4], esi
    inc  lowCount
rsCollectNext:
    inc  esi
    jmp  rsCollect

rsCollected:
    cmp  lowCount, 0
    jne  rsSort
    invoke PutLine, ADDR msgAllGood, CLR_GREEN
    ret

rsSort:                                 ; step 2: bubble sort by medSold, biggest first
    mov  ecx, lowCount
    dec  ecx
    jz   rsPrint
rsOuter:
    push ecx
    xor  esi, esi
rsInner:
    mov  edi, esi
    inc  edi
    cmp  edi, lowCount
    jge  rsInnerDone
    mov  eax, lowList[esi*4]
    mov  ebx, lowList[edi*4]
    mov  edx, medSold[eax*4]
    cmp  edx, medSold[ebx*4]
    jge  rsNoSwap
    mov  lowList[esi*4], ebx
    mov  lowList[edi*4], eax
rsNoSwap:
    inc  esi
    jmp  rsInner
rsInnerDone:
    pop  ecx
    dec  ecx
    jnz  rsOuter

rsPrint:                                ; step 3: print with suggested quantity
    xor  edi, edi
rsPrintLoop:
    cmp  edi, lowCount
    jge  rsEnd
    mov  esi, lowList[edi*4]
    invoke GetUsableStock, esi
    mov  ebx, eax                       ; EBX = usable stock
    mov  eax, medSold[esi*4]
    xor  edx, edx
    mov  ecx, 10
    div  ecx                            ; EAX = sold / 10 (extra for fast sellers)
    mov  ecx, TARGET_STOCK
    sub  ecx, ebx
    add  eax, ecx                       ; order = sold/10 + (30 - usable stock)
    mov  tmpValue, eax
    mov  eax, esi
    call GetNameAddr
    invoke wsprintfA, ADDR lineBuf, ADDR fmtReorder, edx, ebx, medSold[esi*4], tmpValue
    invoke PutLine, ADDR lineBuf, CLR_YELLOW
    inc  edi
    jmp  rsPrintLoop
rsEnd:
    ret
ReorderSuggestion ENDP

; ====================================================================
; 11. SAVE / LOAD FILE
; File = 28 bytes of counters, then the 7 arrays, always in this order.
; ====================================================================

; Writes one block. If Windows reports an error or writes fewer bytes
; than asked, writeFailed is set to 1 so the save can be reported as failed.
WriteChunk PROC pData:DWORD, nBytes:DWORD
    mov  bytesDone, 0
    invoke WriteFile, fileHandle, pData, nBytes, ADDR bytesDone, 0
    test eax, eax
    jz   wcFail
    mov  eax, bytesDone
    cmp  eax, nBytes
    je   wcDone
wcFail:
    mov  writeFailed, 1
wcDone:
    ret
WriteChunk ENDP

; Returns EAX = number of bytes really read
ReadChunk PROC pData:DWORD, nBytes:DWORD
    mov  bytesDone, 0
    invoke ReadFile, fileHandle, pData, nBytes, ADDR bytesDone, 0
    mov  eax, bytesDone
    ret
ReadChunk ENDP

; Writes the file without any message. Returns EAX = 1 ok, 0 failed
SaveQuiet PROC
    mov  writeFailed, 0
    invoke CreateFileA, ADDR dataFileName, GENERIC_WRITE, 0, 0, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, 0
    cmp  eax, INVALID_HANDLE_VALUE
    jne  sqOpened
    xor  eax, eax
    ret
sqOpened:
    mov  fileHandle, eax
    WritePart medCount,  COUNTERS_SIZE
    WritePart medID,     MAX_MEDICINES * 4
    WritePart medName,   MAX_MEDICINES * NAME_SIZE
    WritePart medPrice,  MAX_MEDICINES * 4
    WritePart medSold,   MAX_MEDICINES * 4
    WritePart batMed,    MAX_BATCHES * 4
    WritePart batQty,    MAX_BATCHES * 4
    WritePart batExpiry, MAX_BATCHES * 4
    invoke CloseHandle, fileHandle
    cmp  writeFailed, 0
    jne  sqFail
    mov  eax, 1
    ret
sqFail:
    xor  eax, eax
    ret
SaveQuiet ENDP

; Called after every change so nothing is lost if the window is closed
AutoSave PROC USES eax ecx edx
    call SaveQuiet
    cmp  eax, 1
    je   asOk
    invoke ShowError, ADDR msgSaveError
asOk:
    ret
AutoSave ENDP

SaveAndExit PROC USES eax ecx edx
    call FlushOpenBill
    call SaveQuiet
    cmp  eax, 1
    jne  saeFail
    invoke ShowInfo, ADDR msgSaved
    invoke DestroyWindow, hMain
    ret
saeFail:
    invoke ShowError, ADDR msgSaveError
    ret
SaveAndExit ENDP

; Reads the file if it exists and checks that its contents make sense.
; loadStatus: 0 none, 1 loaded, 2 damaged
LoadDataFromFile PROC USES ebx esi
    mov  loadStatus, 0
    invoke CreateFileA, ADDR dataFileName, GENERIC_READ, 0, 0, OPEN_EXISTING, FILE_ATTRIBUTE_NORMAL, 0
    cmp  eax, INVALID_HANDLE_VALUE
    je   ldDone                         ; no file yet, start empty
    mov  fileHandle, eax
    ReadPart medCount,  COUNTERS_SIZE
    ReadPart medID,     MAX_MEDICINES * 4
    ReadPart medName,   MAX_MEDICINES * NAME_SIZE
    ReadPart medPrice,  MAX_MEDICINES * 4
    ReadPart medSold,   MAX_MEDICINES * 4
    ReadPart batMed,    MAX_BATCHES * 4
    ReadPart batQty,    MAX_BATCHES * 4
    ReadPart batExpiry, MAX_BATCHES * 4
    invoke CloseHandle, fileHandle

    cmp  medCount, MAX_MEDICINES        ; counts must fit the arrays
    ja   ldCorrupt
    cmp  batCount, MAX_BATCHES
    ja   ldCorrupt

    xor  esi, esi                       ; every medicine must look sane
ldCheckMed:
    cmp  esi, medCount
    jge  ldMedOk
    cmp  medID[esi*4], 1
    jb   ldCorrupt
    cmp  medID[esi*4], MAX_ID
    ja   ldCorrupt
    cmp  medPrice[esi*4], 1
    jb   ldCorrupt
    cmp  medPrice[esi*4], MAX_PRICE
    ja   ldCorrupt
    mov  eax, esi                       ; force the last byte of the name to 0
    imul eax, NAME_SIZE
    mov  BYTE PTR medName[eax + NAME_SIZE - 1], 0
    inc  esi
    jmp  ldCheckMed
ldMedOk:

    xor  esi, esi                       ; every batch must point at a real medicine
ldCheckBat:
    cmp  esi, batCount
    jge  ldBatOk
    mov  eax, batMed[esi*4]
    cmp  eax, medCount
    jae  ldCorrupt
    cmp  batQty[esi*4], MAX_QTY
    ja   ldCorrupt
    cmp  batExpiry[esi*4], 20000101
    jb   ldCorrupt
    cmp  batExpiry[esi*4], 21001231
    ja   ldCorrupt
    inc  esi
    jmp  ldCheckBat
ldBatOk:
    mov  loadStatus, 1
    ret
ldReadFail:
    invoke CloseHandle, fileHandle
ldCorrupt:
    mov  medCount, 0
    mov  batCount, 0
    mov  billCount, 0
    mov  itemsSold, 0
    mov  totalSales, 0
    mov  maxBill, 0
    mov  lastSaleDate, 0
    mov  loadStatus, 2
ldDone:
    ret
LoadDataFromFile ENDP
